cmake_minimum_required(VERSION 3.28)
project(HonbakoRuntimeProbe LANGUAGES C CXX OBJC OBJCXX)

set(HONBAKO_RUNTIME_ARTIFACTS "" CACHE PATH "Inspected OpenJDK artifacts outside the repository")
if(NOT EXISTS "${HONBAKO_RUNTIME_ARTIFACTS}/OpenJDK.xcframework/Info.plist")
    message(FATAL_ERROR "Provide the inspected OpenJDK artifacts; no implicit downloads are performed.")
endif()
string(TOLOWER "${CMAKE_OSX_SYSROOT}" HONBAKO_LOWERCASE_SYSROOT)
if(HONBAKO_LOWERCASE_SYSROOT MATCHES "simulator")
    set(OPENJDK_PLATFORM "ios-arm64_x86_64-simulator")
    set(OPENJDK_LIBRARY "libsim.a")
    set(OPENJDK_IMAGE "java_bundle-simulator/lib/modules")
else()
    set(OPENJDK_PLATFORM "ios-arm64")
    set(OPENJDK_LIBRARY "libdevice.a")
    set(OPENJDK_IMAGE "combinedimage/lib/modules")
endif()
set(OPENJDK_PATH "${HONBAKO_RUNTIME_ARTIFACTS}/OpenJDK.xcframework/${OPENJDK_PLATFORM}")
set(HONBAKO_JAVA_RUNTIME_DIAGNOSTICS ON CACHE BOOL "Include isolated proof entry point" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../native" "${CMAKE_CURRENT_BINARY_DIR}/native-executor")
add_executable(HonbakoRuntimeProbe MACOSX_BUNDLE main.mm)
target_compile_features(HonbakoRuntimeProbe PRIVATE cxx_std_17)
target_compile_options(HonbakoRuntimeProbe PRIVATE -fobjc-arc)
target_link_libraries(HonbakoRuntimeProbe PRIVATE HonbakoJavaExtensionExecutor "-framework UIKit")
set_target_properties(HonbakoRuntimeProbe PROPERTIES
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist"
    XCODE_ATTRIBUTE_PRODUCT_BUNDLE_IDENTIFIER "com.missingems.honbaku.runtimeprobe"
    XCODE_ATTRIBUTE_CODE_SIGN_STYLE "Automatic"
    XCODE_ATTRIBUTE_ENABLE_DEBUG_DYLIB "NO"
    XCODE_ATTRIBUTE_TARGETED_DEVICE_FAMILY "1,2"
    XCODE_ATTRIBUTE_IPHONEOS_DEPLOYMENT_TARGET "26.0"
    XCODE_ATTRIBUTE_ENABLE_USER_SCRIPT_SANDBOXING "YES"
)
set(ALL_JAVA_RESOURCES "")
set(CORE_IMAGE "${HONBAKO_RUNTIME_ARTIFACTS}/${OPENJDK_IMAGE}")
if(NOT EXISTS "${CORE_IMAGE}")
    message(FATAL_ERROR "Build the format-1.0 combined mobile runtime image first")
endif()
set_source_files_properties("${CORE_IMAGE}" PROPERTIES MACOSX_PACKAGE_LOCATION "lib/lib" HEADER_FILE_ONLY TRUE)
list(APPEND ALL_JAVA_RESOURCES "${CORE_IMAGE}")
set(TIMEZONE_DATA "${HONBAKO_RUNTIME_ARTIFACTS}/java-data/tzdb.dat")
set_source_files_properties("${TIMEZONE_DATA}" PROPERTIES MACOSX_PACKAGE_LOCATION "lib/lib" HEADER_FILE_ONLY TRUE)
list(APPEND ALL_JAVA_RESOURCES "${TIMEZONE_DATA}")
set(HONBAKO_BRIDGE_JAR "${HONBAKO_RUNTIME_ARTIFACTS}/honbako-extension-bridge-v2.jar" CACHE FILEPATH "Verified Java bridge build")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bridge-resource")
configure_file("${HONBAKO_BRIDGE_JAR}" "${CMAKE_CURRENT_BINARY_DIR}/bridge-resource/honbako-extension-bridge.jar" COPYONLY)
set(BRIDGE_RESOURCE "${CMAKE_CURRENT_BINARY_DIR}/bridge-resource/honbako-extension-bridge.jar")
set_source_files_properties("${BRIDGE_RESOURCE}" PROPERTIES MACOSX_PACKAGE_LOCATION "." HEADER_FILE_ONLY TRUE)
list(APPEND ALL_JAVA_RESOURCES "${BRIDGE_RESOURCE}")
foreach(JAR IN ITEMS RuntimeProbe.jar komga-proof-dependencies.jar tachiyomi-all.komga-v1.6.74.jar)
    set(RESOURCE "${HONBAKO_RUNTIME_ARTIFACTS}/${JAR}")
    set_source_files_properties("${RESOURCE}" PROPERTIES MACOSX_PACKAGE_LOCATION "." HEADER_FILE_ONLY TRUE)
    list(APPEND ALL_JAVA_RESOURCES "${RESOURCE}")
endforeach()
foreach(PAGE IN ITEMS cover.png page-1.png page-2.png page-3.png)
    set(RESOURCE "${HONBAKO_RUNTIME_ARTIFACTS}/fixture/${PAGE}")
    set_source_files_properties("${RESOURCE}" PROPERTIES MACOSX_PACKAGE_LOCATION "fixture" HEADER_FILE_ONLY TRUE)
    list(APPEND ALL_JAVA_RESOURCES "${RESOURCE}")
endforeach()
target_sources(HonbakoRuntimeProbe PRIVATE ${ALL_JAVA_RESOURCES})

file(GLOB_RECURSE PROBE_CONF_RESOURCES CONFIGURE_DEPENDS "${HONBAKO_RUNTIME_ARTIFACTS}/java-conf/*")
foreach(RESOURCE IN LISTS PROBE_CONF_RESOURCES)
    file(RELATIVE_PATH CONF_PATH "${HONBAKO_RUNTIME_ARTIFACTS}/java-conf" "${RESOURCE}")
    get_filename_component(CONF_FOLDER "${CONF_PATH}" DIRECTORY)
    set_source_files_properties("${RESOURCE}" PROPERTIES MACOSX_PACKAGE_LOCATION "lib/conf/${CONF_FOLDER}" HEADER_FILE_ONLY TRUE)
endforeach()
target_sources(HonbakoRuntimeProbe PRIVATE ${PROBE_CONF_RESOURCES})
